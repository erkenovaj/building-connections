# Training-only ML stack. Install into a dedicated venv (e.g. .venv-train);
# do NOT add these to pyproject.toml — that is the serverless deploy manifest
# and these multi-GB deps would bloat the Vercel bundle.
#
# The trainer imports the app code (connections_sampler -> ortools), so pull in
# the app runtime manifest too; this file is the single install target on Kaggle.
-r requirements.txt
torch==2.10.0
# torchvision must match torch's ABI (transformers.image_utils imports it);
# platforms preinstall a torchvision built for a different torch, which
# breaks with "operator torchvision::nms does not exist". 0.25.x pairs
# with torch 2.10.
torchvision==0.25.0
transformers>=5.12.0
trl>=1.7.0
datasets>=5.0.0
accelerate>=1.14.0
peft>=0.19.0
# Kaggle's base image preinstalls torchao 0.10.0, which trips peft's LoRA
# version gate (needs >0.16.0) and aborts get_peft_model. We don't use torchao
# quant; pin 0.17.0 — clears the gate and keeps torch on 2.10.0 (its cpp-ext
# skip warning is harmless since those kernels are never called).
torchao==0.17.0
